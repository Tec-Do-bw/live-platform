# 登录转发投屏技术设计方案

## 1. 目标与范围

### 1.1 目标
- 在 AdsPower 创建的浏览器环境中完成登录，并实时投屏给前端
- 前端可低延迟查看登录过程
- 登录成功后完成校验与回传

### 1.2 范围
- 基于原生 CDP WebSocket 的投屏链路
- DrissionPage 负责页面导航和登录监听
- FastAPI 服务：创建环境、启动浏览器、建立 WS、关闭浏览器

### 1.3 非目标
- 账号登录逻辑自动化（用户手动操作）
- 前端 UI 细节
- 持久化存储设计（由业务后端处理）

---

## 2. 技术栈

| 组件 | 技术选型 | 用途 |
|------|----------|------|
| Web 框架 | FastAPI | HTTP/WebSocket 接口 |
| 浏览器管理 | AdsPower API | 创建/启动/关闭浏览器环境 |
| 页面操作 | DrissionPage | 导航到登录页、监听登录成功 |
| 投屏转发 | 原生 CDP WebSocket | screencast 帧转发（低延迟） |
| 异步框架 | asyncio + websockets | CDP 连接和帧转发 |

## 4. 关键技术决策

### 4.1 为什么投屏用原生 CDP 而不是 DrissionPage？

| 对比项 | DrissionPage | 原生 CDP WebSocket |
|--------|--------------|-------------------|
| 帧接收方式 | 内置 screencast 录制功能 | 直接监听 screencastFrame 事件 |
| 实时转发 | 需要轮询或适配 | 事件驱动，天然适合实时转发 |
| 延迟 | 较高（多一层封装） | 更低 |
| 复杂度 | 简单 | 需要自己实现事件监听 |

**结论**：投屏需要实时帧转发，原生 CDP WebSocket 的事件驱动模式更适合。

### 4.2 为什么登录监听用 DrissionPage？

DrissionPage 内置了 `tab.listen` 网络监听功能：
- 可以监听特定 URL 的响应
- 支持超时等待
- 自动解析响应体
- 无需自己实现 CDP Network 事件处理

**示例代码**：
```python
tab.listen.start('api/user/info')  # 监听用户信息接口
res = tab.listen.wait(timeout=900)  # 等待最多15分钟
if res:
    user_info = res.response.body  # 获取响应内容
```

### 4.3 DrissionPage 和原生 CDP 能同时连接吗？

**可以**。两者连接到同一个 Chrome 调试端口，互不冲突：
- DrissionPage 通过 `Chromium(addr_or_opts=(ip, port))` 连接
- 原生 CDP 通过 `websockets.connect(ws_url)` 连接

```python
# services/login_monitor.py

class ElementLoginMonitor:
    """基于页面元素的登录检测"""

    # TODO: 后续补充具体元素选择器
    SELECTORS = {
        "tiktok": {
            "avatar": "",      # 用户头像选择器
            "nickname": "",    # 昵称选择器
            "logged_in": "",   # 登录成功标识元素
        },
        "shopee": {
            "avatar": "",
            "nickname": "",
            "logged_in": "",
        }
    }

    def start(self, session: Session, timeout: int = 900):
        tab = session.drissionpage_tab
        selectors = self.SELECTORS.get(session.media, {})

        if not selectors.get("logged_in"):
            return None

        start_time = time.time()

        # 轮询检测
        while time.time() - start_time < timeout:
            try:
                # 检测登录成功标识元素
                ele = tab.ele(selectors["logged_in"], timeout=5)
                if ele:
                    # 提取用户信息
                    return self._extract_user_info(tab, selectors)
            except Exception:
                pass

            time.sleep(2)  # 每2秒检测一次

        return None

    def _extract_user_info(self, tab, selectors: dict) -> dict:
        """从页面元素提取用户信息"""
        info = {}

        if selectors.get("nickname"):
            ele = tab.ele(selectors["nickname"], timeout=3)
            if ele:
                info["nickname"] = ele.text

        # TODO: 提取其他信息

        return info
```

#### 混合监听器

```python
# services/login_monitor.py

class HybridLoginMonitor:
    """混合登录监听器 - 同时使用网络监听和元素检测"""

    def __init__(self):
        self.network_monitor = NetworkLoginMonitor()
        self.element_monitor = ElementLoginMonitor()

    async def start(self, session: Session) -> Optional[dict]:
        """启动混合监听"""
        import asyncio
        from concurrent.futures import ThreadPoolExecutor

        with ThreadPoolExecutor(max_workers=2) as executor:
            loop = asyncio.get_event_loop()

            # 并行启动两种监听
            tasks = [
                loop.run_in_executor(executor, self.network_monitor.start, session),
                loop.run_in_executor(executor, self.element_monitor.start, session, 900),
            ]

            # 等待任意一个完成
            done, pending = await asyncio.wait(
                tasks,
                return_when=asyncio.FIRST_COMPLETED
            )

            # 取消未完成的任务
            for task in pending:
                task.cancel()

            # 返回第一个成功的结果
            for task in done:
                result = task.result()
                if result:
                    return result

        return None

    async def on_login_success(self, session: Session, login_info: dict):
        """登录成功回调"""
        # 1. 校验登录信息
        if not self._validate(session, login_info):
            return False

        # 2. 通知前端
        await session.send_to_frontend({
            "type": "login_success",
            "data": login_info
        })

        # 3. 回传业务后端
        await self._callback_backend(session, login_info)

        # 4. 移动 AdsPower 环境到用户组
        await self._move_to_group(session)

        return True
```

### 8.2 CDP 投屏转发 (`services/screencast.py`)

```python
class ScreencastService:
    async def start(self, session: Session, frontend_ws: WebSocket):
        """启动投屏转发"""
        # 1. 连接到 Chrome CDP
        cdp_ws = await websockets.connect(session.ws_url)

        # 2. 启动 screencast
        await cdp_ws.send(json.dumps({
            "id": 1,
            "method": "Page.startScreencast",
            "params": {
                "format": "jpeg",
                "quality": 80,
                "maxWidth": 1920,
                "maxHeight": 1080
            }
        }))

        # 3. 事件循环：接收帧并转发
        async for msg in cdp_ws:
            data = json.loads(msg)
            if data.get("method") == "Page.screencastFrame":
                # 转发帧到前端
                await frontend_ws.send_json({
                    "type": "frame",
                    "data": data["params"]["data"],
                    "width": data["params"]["metadata"]["width"],
                    "height": data["params"]["metadata"]["height"],
                    "ts": time.time()
                })

                # 确认帧
                await cdp_ws.send(json.dumps({
                    "id": 2,
                    "method": "Page.screencastFrameAck",
                    "params": {"sessionId": data["params"]["sessionId"]}
                }))
```

### 8.3 输入事件转发

```python
async def handle_input(self, cdp_ws, event: dict):
    """将前端输入事件转发到 Chrome"""
    if event["event"] == "mouse":
        await cdp_ws.send(json.dumps({
            "id": 3,
            "method": "Input.dispatchMouseEvent",
            "params": {
                "type": event["action"],  # mousePressed/mouseReleased/mouseMoved
                "x": event["x"],
                "y": event["y"],
                "button": event["button"]
            }
        }))
    elif event["event"] == "key":
        await cdp_ws.send(json.dumps({
            "id": 4,
            "method": "Input.dispatchKeyEvent",
            "params": {
                "type": "keyDown" if event["action"] == "down" else "keyUp",
                "key": event["key"]
            }
        }))
```
